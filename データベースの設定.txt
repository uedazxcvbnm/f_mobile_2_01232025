set names utf8;
drop database if exists pbl2;
create database pbl2 character set utf8 collate utf8_general_ci;
# ユーザー「kobe」にパスワード「denshi」を設定し、データベース「pbl2」に対する全ての権限を付与
grant all privileges on pbl2.* to kobe@localhost identified by 'denshi';
# データベースpbl2 を使用
use pbl2;


CREATE TABLE drink (
  drink_id int(11) auto_increment primary key,
  drink_name varchar(50),
  alc int(11) NOT NULL
);

# テーブル daily_record の作成
# sBP：最高血圧　dBP：最低血圧
create table daily_record(ident int auto_increment primary key,
 drink_id INT,
 glass_id INT,                         
 FOREIGN KEY (drink_id) REFERENCES drink(drink_id),
 FOREIGN KEY (glass_id) REFERENCES glass(glass_id),
 date date,
 account_drunk INT,
 money INT,
 sBP INT,
 dBP INT,
 alchol_volume int(11) NOT NULL
 dBP INT
);

#テーブル teams の作成
create table teams(ident int auto_increment primary key,
    name character(20),
    detail character(200),
    size int
);
