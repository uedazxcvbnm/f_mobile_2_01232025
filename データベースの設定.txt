# テーブル daily_record の作成
# sBP：最高血圧　dBP：最低血圧
create table daily_record(ident int auto_increment primary key,
 date date,
 money INT,
 alchol_volume int(11) NOT NULL,
 user_id INT,
 FOREIGN KEY (user_id) REFERENCES user(user_id)
);

create table record_oneday(ident int auto_increment primary key,
 drink_id INT,
 glass_id INT,                         
 FOREIGN KEY (drink_id) REFERENCES drink(drink_id),
 FOREIGN KEY (glass_id) REFERENCES glass(glass_id),
 date date,
 account_drunk INT,
 money INT,
 alchol_volume int(11) NOT NULL,
 user_id INT,
 FOREIGN KEY (user_id) REFERENCES user(user_id)
);

# chatテーブル(#3)
CREATE TABLE IF NOT EXISTS chat (
    id INT AUTO_INCREMENT PRIMARY KEY,
    message TEXT NOT NULL,
    date DATETIME NOT NULL,
    user_id INT,
    FOREIGN KEY (user_id) REFERENCES user(user_id)
);

# テーブル posts の作成
CREATE TABLE posts (
  id INT AUTO_INCREMENT PRIMARY KEY,
  post_title VARCHAR(255),
  post_content TEXT,
  post_url VARCHAR(255),
  post_image VARCHAR(255),
  post_tags VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  likes INT DEFAULT 0
);



# デーブル　post comment の作成
CREATE TABLE comments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    post_id INT,
    comment_text TEXT,
    likes INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(id)
);
