-- daily_record テーブル
CREATE TABLE daily_record (
  ident INT AUTO_INCREMENT PRIMARY KEY,
  date DATE,
  money INT,
  alchol_volume INT NOT NULL,
  user_id INT,
  FOREIGN KEY (user_id) REFERENCES user(user_id)
);

-- record_oneday テーブル
CREATE TABLE record_oneday (
  ident INT AUTO_INCREMENT PRIMARY KEY,
  drink_id INT,
  glass_id INT,                         
  FOREIGN KEY (drink_id) REFERENCES drink(drink_id),
  FOREIGN KEY (glass_id) REFERENCES glass(glass_id),
  date DATE,
  account_drunk INT,
  money INT,
  alchol_volume INT NOT NULL,
  user_id INT,
  FOREIGN KEY (user_id) REFERENCES user(user_id)
);

-- chat テーブル (#3)
CREATE TABLE IF NOT EXISTS chat (
    id INT AUTO_INCREMENT PRIMARY KEY,
    message TEXT NOT NULL,
    date DATETIME NOT NULL,
    user_id INT,
    FOREIGN KEY (user_id) REFERENCES user(user_id)
);

-- posts テーブル
CREATE TABLE posts (
  id INT AUTO_INCREMENT PRIMARY KEY,
  post_title VARCHAR(255),
  post_content TEXT,
  post_url VARCHAR(255),
  post_tags VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  likes INT DEFAULT 0,
  user_id INT,
  FOREIGN KEY (user_id) REFERENCES user(user_id)
);

-- comments テーブル
CREATE TABLE comments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    post_id INT,
    comment_text TEXT,
    likes INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    user_id INT,
    FOREIGN KEY (post_id) REFERENCES posts(id),
    FOREIGN KEY (user_id) REFERENCES user(user_id)
);

-- likes テーブル
CREATE TABLE likes (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT,
  post_id INT,
  UNIQUE KEY unique_like (user_id, post_id),  
  FOREIGN KEY (user_id) REFERENCES user(user_id),
  FOREIGN KEY (post_id) REFERENCES posts(id)
);
